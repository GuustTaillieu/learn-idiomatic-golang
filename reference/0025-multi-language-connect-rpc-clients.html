Reference: Multi-Language Connect-RPC Clients

How modern multi-platform architectures consume a Go Connect-RPC backend from TypeScript (Web), Rust (Systems), and Kotlin (Android Jetpack Compose).

The Multi-Platform Advantage:
"A single .proto contract generates type-safe clients across languages without writing duplicate data models. Connect-RPC endpoints simultaneously accept binary Protobuf and standard HTTP JSON on the exact same port."

1. TypeScript (Web / React / Next.js)

Using @connectrpc/connect and @connectrpc/connect-web:

// 1. Install dependencies:
// npm install @connectrpc/connect @connectrpc/connect-web @bufbuild/protobuf

import { createClient } from "@connectrpc/connect";
import { createConnectTransport } from "@connectrpc/connect-web";
import { OrderService } from "./gen/order/v1/order_connect";

// 2. Initialize transport (points directly to Go backend, no Envoy proxy needed!)
const transport = createConnectTransport({
  baseUrl: "http://localhost:8080",
});

const client = createClient(OrderService, transport);

// 3. Make 100% type-safe RPC call
export async function createOrder(itemId: string, amount: number) {
  try {
    const res = await client.createOrder({
      itemId: itemId,
      amount: amount,
    });
    console.log("Order Created:", res.orderId, "Status:", res.status);
    return res;
  } catch (err) {
    console.error("RPC Error:", err);
    throw err;
  }
}

2. Kotlin (Android / Jetpack Compose)

Using the official Connect-Kotlin SDK (com.connectrpc:connect-kotlin):

// build.gradle.kts:
// implementation("com.connectrpc:connect-kotlin:0.7.1")
// implementation("com.connectrpc:connect-kotlin-okhttp:0.7.1")

import com.connectrpc.ProtocolClientConfig
import com.connectrpc.okhttp.ConnectOkHttpClient
import com.connectrpc.ProtocolClient
import okhttp3.OkHttpClient
import order.v1.OrderServiceClient
import order.v1.CreateOrderRequest

class OrderRepository {
    // 1. Create client using OkHttp
    private val okHttpClient = OkHttpClient.Builder().build()
    private val protocolClient = ProtocolClient(
        httpClient = ConnectOkHttpClient(okHttpClient),
        config = ProtocolClientConfig(
            host = "http://10.0.2.2:8080", // 10.0.2.2 maps to host localhost from Android Emulator
        )
    )
    private val client = OrderServiceClient(protocolClient)

    // 2. Suspending function for Jetpack Compose ViewModel
    suspend fun submitOrder(itemId: String, amount: Int): Result<String> {
        val req = CreateOrderRequest(itemId = itemId, amount = amount)
        val response = client.createOrder(req)
        
        return response.fold(
            onSuccess = { res -> Result.success(res.orderId) },
            onFailure = { err -> Result.failure(err) }
        )
    }
}

// 3. Jetpack Compose UI
@Composable
fun OrderButton(viewModel: OrderViewModel) {
    val coroutineScope = rememberCoroutineScope()
    Button(onClick = {
        coroutineScope.launch {
            viewModel.submitOrder("00000000-0000-0000-0000-000000000001", 1)
        }
    }) {
        Text("⚡ Dispatch to Go Backend")
    }
}

3. Rust

In Rust, you can consume Connect-RPC either with type-safe generated structs or via high-performance HTTP JSON:

// Cargo.toml:
// reqwest = { version = "0.12", features = ["json"] }
// serde = { version = "1.0", features = ["derive"] }
// tokio = { version = "1", features = ["full"] }

use serde::{Deserialize, Serialize};

#[derive(Serialize)]
struct CreateOrderRequest {
    item_id: String,
    amount: i32,
}

#[derive(Deserialize, Debug)]
struct CreateOrderResponse {
    order_id: String,
    status: String,
}

#[tokio::main]
async fn main() -> Result<(), Box<dyn std::error::Error>> {
    let client = reqwest::Client::new();

    let req = CreateOrderRequest {
        item_id: "00000000-0000-0000-0000-000000000001".to_string(),
        amount: 2,
    };

    // Standard Connect POST endpoint
    let res: CreateOrderResponse = client
        .post("http://localhost:8080/order.v1.OrderService/CreateOrder")
        .header("Content-Type", "application/json")
        .json(&req)
        .send()
        .await?
        .json()
        .await?;

    println!("Created Order from Rust: ID = {}, Status = {}", res.order_id, res.status);
    Ok(())
}
